local function tween(object, duration, props, style)
	local t = TweenService:Create(
		object,
		TweenInfo.new(
			duration or 0.2,
			style or Enum.EasingStyle.Quad,
			Enum.EasingDirection.Out
		),
		props
	)
	t:Play()
	return t
end

local function corner(object, radius)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(0, radius)
	c.Parent = object
	return c
end

local function stroke(object, color, thickness, transparency)
	local s = Instance.new("UIStroke")
	s.Color = color
	s.Thickness = thickness or 1
	s.Transparency = transparency or 0
	s.Parent = object
	return s
end

local function gradient(object, c1, c2, rotation)
	local g = Instance.new("UIGradient")
	g.Color = ColorSequence.new({
		ColorSequenceKeypoint.new(0, c1),
		ColorSequenceKeypoint.new(1, c2)
	})
	g.Rotation = rotation or 0
	g.Parent = object
	return g
end

local speedEnabled = false
local speedValue = 50
local speedConnections = {}

local function applySpeed()
	local character = player.Character
	if not character then
		return
	end

	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if not humanoid then
		return
	end

	humanoid.WalkSpeed = speedEnabled and speedValue or 16

	if speedConnections[humanoid] then
		speedConnections[humanoid]:Disconnect()
	end

	speedConnections[humanoid] = humanoid:GetPropertyChangedSignal("WalkSpeed"):Connect(function()
		if speedEnabled and humanoid.WalkSpeed ~= speedValue then
			humanoid.WalkSpeed = speedValue
		end
	end)
end

local function setSpeedEnabled(enabled)
	speedEnabled = enabled
	applySpeed()
end

player.CharacterAdded:Connect(function()
	task.wait(0.5)
	applySpeed()
end)

local glassOriginals = {}
local glassParts = {}
local glassLoop = false

local BODY_PART_NAMES = {
	Head = true,
	UpperTorso = true,
	LowerTorso = true,
	LeftUpperArm = true,
	LeftLowerArm = true,
	LeftHand = true,
	RightUpperArm = true,
	RightLowerArm = true,
	RightHand = true,
	LeftUpperLeg = true,
	LeftLowerLeg = true,
	LeftFoot = true,
	RightUpperLeg = true,
	RightLowerLeg = true,
	RightFoot = true,
	Torso = true,
	["Left Arm"] = true,
	["Right Arm"] = true,
	["Left Leg"] = true,
	["Right Leg"] = true
}

local function isRealBodyPart(obj)
	if not obj:IsA("BasePart") then
		return false
	end
	if obj.Name == "HumanoidRootPart" then
		return false
	end
	if not BODY_PART_NAMES[obj.Name] then
		return false
	end
	if obj:FindFirstAncestorOfClass("Accessory") then
		return false
	end
	if obj:FindFirstAncestorOfClass("Tool") then
		return false
	end
	return true
end

local function restoreGlass()
	glassLoop = false

	for part, original in pairs(glassOriginals) do
		if part and part.Parent then
			pcall(function()
				part.Material = original.Material
				part.Color = original.Color
				part.Transparency = original.Transparency
				part.Reflectance = original.Reflectance
			end)
		end
	end

	table.clear(glassOriginals)
	table.clear(glassParts)
end

local function enableGlass()
	local character = player.Character
	if not character then
		return
	end

	restoreGlass()
	glassBodyEnabled = true
	glassLoop = true

	for _, obj in ipairs(character:GetDescendants()) do
		if isRealBodyPart(obj) then
			glassOriginals[obj] = {
				Material = obj.Material,
				Color = obj.Color,
				Transparency = obj.Transparency,
				Reflectance = obj.Reflectance
			}

			table.insert(glassParts, obj)

			obj.Material = Enum.Material.Glass
			obj.Transparency = 0.32
			obj.Reflectance = 0.05
		end
	end

	task.spawn(function()
		while glassLoop and glassBodyEnabled and player.Character == character do
			local t = os.clock()
			local wave = (math.sin(t * 1.4) + 1) / 2

			local color = Color3.new(
				0.45 + wave * 0.25,
				0.10 + wave * 0.13,
				0.72 + wave * 0.24
			)

			for _, part in ipairs(glassParts) do
				if part and part.Parent then
					part.Color = color
				end
			end

			task.wait(0.05)
		end
	end)
end

local function setGlassBody(enabled)
	glassBodyEnabled = enabled

	if enabled then
		enableGlass()
	else
		restoreGlass()
	end
end

player.CharacterAdded:Connect(function()
	task.wait(1)
	if glassBodyEnabled then
		enableGlass()
	end
end)

local function teleportAmethystsToPlayer()
	local character = player.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")
	local rooms = workspace:FindFirstChild("CurrentRooms")

	if not root or not rooms then
		return
	end

	for _, room in ipairs(rooms:GetChildren()) do
		for _, obj in ipairs(room:GetDescendants()) do
			if obj.Name == "SingleAmethystShard" then
				pcall(function()
					if obj:IsA("Model") then
						obj:PivotTo(root.CFrame)
					elseif obj:IsA("BasePart") then
						obj.CFrame = root.CFrame
					end
				end)
			end
		end
	end
end

local function findKeycard(room)
	for _, obj in ipairs(room:GetDescendants()) do
		if obj.Name == "Keycard" then

end

local function findDoor(room)
	return room:FindFirstChild("Door")
end

local function findLever(room)
	local scriptables = room:FindFirstChild("Scriptables")
	if scriptables then
		return scriptables:FindFirstChild("MainLever")
	end
	return nil
end

local FINDERS = {
	Keycard = findKeycard,
	Door = findDoor,
	Lever = findLever
}

local AMETHYST_NAMES = {
	MidAmethyst = true,
	SmallAmethyst = true,
	BigAmethyst = true,
	SingleAmethystShard3 = true
}

local function findAmethysts(room)
	local result = {}

	for _, obj in ipairs(room:GetDescendants()) do
		if AMETHYST_NAMES[obj.Name] then
			table.insert(result, obj)
		end
	end

	return result
end

local function applyESP(obj, category)
	if not obj or not obj.Parent then
		return
	end

	local settings = CONFIG[category]
	local hName = category .. "Highlight"
	local bName = category .. "Label"

	local highlight = obj:FindFirstChild(hName)

	if not highlight then
		highlight = Instance.new("Highlight")
		highlight.Name = hName
		highlight.FillColor = settings.color
		highlight.OutlineColor = settings.color
		highlight.FillTransparency = 0.5
		highlight.OutlineTransparency = 0
		highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
		highlight.Parent = obj
	end

	local billboard = obj:FindFirstChild(bName)

	if not billboard then
		billboard = Instance.new("BillboardGui")
		billboard.Name = bName
		billboard.Size = UDim2.fromOffset(120, 32)
		billboard.StudsOffset = Vector3.new(0, 2.2, 0)
		billboard.AlwaysOnTop = true
		billboard.MaxDistance = 1000
		billboard.Parent = obj

		local text = Instance.new("TextLabel")
		text.Name = "Text"
		text.Size = UDim2.fromScale(1, 1)
		text.BackgroundTransparency = 1
		text.Text = settings.label
		text.TextColor3 = settings.color
		text.TextStrokeTransparency = 0
		text.TextStrokeColor3 = Color3.new(0, 0, 0)
		text.Font = Enum.Font.GothamBold
		text.TextScaled = true
		text.Parent = billboard
	end
end

local function removeESP(obj, category)
	if not obj then
		return
	end

	local h = obj:FindFirstChild(category .. "Highlight")
	local l = obj:FindFirstChild(category .. "Label")

	if h then h:Destroy() end
	if l then l:Destroy() end
end

local function processRoom(room)
	if not room then
		return
	end

	for category, finder in pairs(FINDERS) do
		local obj = finder(room)

		if obj then
			if CONFIG[category].enabled then
				applyESP(obj, category)
			else
				removeESP(obj, category)
			end
		end
	end

	for _, obj in ipairs(findAmethysts(room)) do
		if CONFIG.Amethyst.enabled then
			applyESP(obj, "Amethyst")
		else
			removeESP(obj, "Amethyst")
		end
	end
end

local function scanRooms()
	local folder = workspace:FindFirstChild("CurrentRooms")
	if not folder then
		return
	end

	for _, room in ipairs(folder:GetChildren()) do
		pcall(function()
			processRoom(room)
		end)
	end
end

local function clearCategory(category)
	local hName = category .. "Highlight"
	local lName = category .. "Label"

	for _, obj in ipairs(workspace:GetDescendants()) do
		if obj.Name == hName or obj.Name == lName then
			obj:Destroy()
		end
	end
end

task.spawn(function()
	while task.wait(1) do
		pcall(scanRooms)
	end
end)

task.spawn(function()
	local folder = workspace:WaitForChild("CurrentRooms", 15)

	if folder then
		folder.ChildAdded:Connect(function(room)
			task.wait(0.35)
			pcall(function()
				processRoom(room)
			end)
		end)
	end
end)

local function findMonsterFolder()
	for _, name in ipairs(MONSTER_FOLDER_NAMES) do
		local folder = workspace:FindFirstChild(name)
		if folder then
			return folder
		end
	end

	for _, obj in ipairs(workspace:GetDescendants()) do
		if obj:IsA("Folder") or obj:IsA("Model") then
			local lower = string.lower(obj.Name)

			if lower == "monsters"
				or lower == "entities"
				or lower == "monster" then
				return obj
			end
		end
	end

	return nil
end

local monitoredFolders = {}

local function monitorMonsterFolder(folder)
	if not folder or monitoredFolders[folder] then
		return
	end

	monitoredFolders[folder] = true

	folder.ChildAdded:Connect(function(obj)
		if not monsterNotificationsEnabled then
			return
		end

		task.wait(0.05)

		if obj and obj.Parent == folder then
			showNotification(obj.Name)
		end
	end)
end

task.spawn(function()
	while task.wait(1) do
		local folder = findMonsterFolder()
		if folder then
			monitorMonsterFolder(folder)
		end
	end
end)

local function makeToggle(text, description, callback)
	local holder = Instance.new("Frame")

	nameLabel.Text = text
	nameLabel.TextColor3 = Color3.fromRGB(245, 243, 250)
	nameLabel.Font = Enum.Font.GothamBold
	nameLabel.TextSize = 13
	nameLabel.TextXAlignment = Enum.TextXAlignment.Left
	nameLabel.ZIndex = 13
	nameLabel.Parent = holder

	local descLabel = Instance.new("TextLabel")
	descLabel.BackgroundTransparency = 1
	descLabel.Position = UDim2.fromOffset(14, 30)
	descLabel.Size = UDim2.new(1, -82, 0, 17)
	descLabel.Text = description
	descLabel.TextColor3 = Color3.fromRGB(130, 125, 145)
	descLabel.Font = Enum.Font.Gotham
	descLabel.TextSize = 8
	descLabel.TextXAlignment = Enum.TextXAlignment.Left
	descLabel.TextTruncate = Enum.TextTruncate.AtEnd
	descLabel.ZIndex = 13
	descLabel.Parent = holder

	local button = Instance.new("TextButton")
	button.Size = UDim2.fromOffset(48, 26)
	button.Position = UDim2.new(1, -61, 0.5, -13)
	button.BackgroundColor3 = Color3.fromRGB(48, 44, 60)
	button.BorderSizePixel = 0
	button.Text = ""
	button.AutoButtonColor = false
	button.ZIndex = 13
	button.Parent = holder
	corner(button, 13)

	local knob = Instance.new("Frame")
	knob.Size = UDim2.fromOffset(20, 20)
	knob.Position = UDim2.new(0, 3, 0.5, -10)
	knob.BackgroundColor3 = Color3.fromRGB(185, 180, 195)
	knob.BorderSizePixel = 0
	knob.ZIndex = 14
	knob.Parent = button
	corner(knob, 10)

	local state = false

	local function update()
		if state then
			tween(button, 0.16, {BackgroundColor3 = Color3.fromRGB(110, 55, 230)})
			tween(knob, 0.16, {
				Position = UDim2.new(1, -23, 0.5, -10),
				BackgroundColor3 = Color3.fromRGB(255, 255, 255)
			})
		else
			tween(button, 0.16, {BackgroundColor3 = Color3.fromRGB(48, 44, 60)})
			tween(knob, 0.16, {
				Position = UDim2.new(0, 3, 0.5, -10),
				BackgroundColor3 = Color3.fromRGB(185, 180, 195)
			})
		end

		pcall(callback, state)
	end

	button.MouseButton1Click:Connect(function()
		state = not state
		update()
	end)

	return holder, function(value)
		state = value
		update()
	end
end

local fullSize = UDim2.fromOffset(360, 530)
local miniSize = UDim2.fromOffset(360, 82)

local function setMinimized(value)
	minimized = value

	if minimized then
		content.Visible = false
		reopen.Visible = true

		tween(main, 0.22, {
			Size = miniSize
		})

		tween(reopen, 0.22, {
			Size = UDim2.fromOffset(48, 48)
		})
	else
		reopen.Visible = false
		content.Visible = true

		tween(main, 0.22, {
			Size = fullSize
		})
	end
end

minimize.MouseButton1Click:Connect(function()
	setMinimized(true)
end)

reopen.MouseButton1Click:Connect(function()
	setMinimized(false)
end)

--==============================================================
-- DRAGGING
--==============================================================

local dragging = false
local dragStart
local startPos

header.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then

		dragging = true
		dragStart = input.Position
		startPos = main.Position

		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if not dragging then
		return
	end

	if input.UserInputType == Enum.UserInputType.MouseMovement
		or input.UserInputType == Enum.UserInputType.Touch then

		local delta = input.Position - dragStart

		main.Position = UDim2.new(
			startPos.X.Scale,
			startPos.X.Offset + delta.X,
			startPos.Y.Scale,
			startPos.Y.Offset + delta.Y
		)
	end
end)
