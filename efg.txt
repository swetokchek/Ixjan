local function tween(object, duration, properties, style)
	local info = TweenInfo.new(
		duration or 0.2,
		style or Enum.EasingStyle.Quad,
		Enum.EasingDirection.Out
	)

	return TweenService:Create(object, info, properties)
end

local function addCorner(object, radius)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(0, radius)
	c.Parent = object
	return c
end

local function addStroke(object, color, thickness, transparency)
	local s = Instance.new("UIStroke")
	s.Color = color
	s.Thickness = thickness or 1
	s.Transparency = transparency or 0
	s.Parent = object
	return s
end

local function addGradient(object, c1, c2, rotation)
	local g = Instance.new("UIGradient")

	g.Color = ColorSequence.new({
		ColorSequenceKeypoint.new(0, c1),
		ColorSequenceKeypoint.new(1, c2)
	})

	g.Rotation = rotation or 0
	g.Parent = object

	return g
end

local glassOriginals = {}
local glassParts = {}
local glassLoop = false

local BODY_PART_NAMES = {

	Head = true,

	UpperTorso = true,
	LowerTorso = true,

	LeftUpperArm = true,
	LeftLowerArm = true,
	LeftHand = true,

	RightUpperArm = true,
	RightLowerArm = true,
	RightHand = true,

	LeftUpperLeg = true,
	LeftLowerLeg = true,
	LeftFoot = true,

	RightUpperLeg = true,
	RightLowerLeg = true,
	RightFoot = true,

	Torso = true,
	["Left Arm"] = true,
	["Right Arm"] = true,
	["Left Leg"] = true,
	["Right Leg"] = true
}

local function restoreGlass()

	glassLoop = false

	for part, original in pairs(glassOriginals) do

		if part and part.Parent then

			pcall(function()

				part.Material = original.Material
				part.Color = original.Color
				part.Transparency = original.Transparency
				part.Reflectance = original.Reflectance

			end)

		end
	end

	table.clear(glassOriginals)
	table.clear(glassParts)
end

local function isRealBodyPart(obj)

	if not obj:IsA("BasePart") then
		return false
	end

	if obj.Name == "HumanoidRootPart" then
		return false
	end

	if not BODY_PART_NAMES[obj.Name] then
		return false
	end

	if obj:FindFirstAncestorOfClass("Accessory") then
		return false
	end

	if obj:FindFirstAncestorOfClass("Tool") then
		return false
	end

	return true
end

local function enableGlass()

	local character = player.Character

	if not character then
		return
	end

	restoreGlass()

	glassBodyEnabled = true
	glassLoop = true

	for _, obj in ipairs(character:GetDescendants()) do

		if isRealBodyPart(obj) then

			glassOriginals[obj] = {
				Material = obj.Material,
				Color = obj.Color,
				Transparency = obj.Transparency,
				Reflectance = obj.Reflectance
			}

			table.insert(glassParts, obj)

			obj.Material = Enum.Material.Glass
			obj.Transparency = 0.32
			obj.Reflectance = 0.05

		end
	end

	task.spawn(function()

		while glassLoop
			and glassBodyEnabled
			and player.Character == character do

			local t = os.clock()

			local wave =
				(math.sin(t * 1.4) + 1) / 2

			local color = Color3.new(
				0.45 + wave * 0.25,
				0.10 + wave * 0.13,
				0.72 + wave * 0.24
			)

			for _, part in ipairs(glassParts) do

				if part and part.Parent then
					part.Color = color
				end

			end

			task.wait(0.05)

		end
	end)
end

local function setGlassBody(enabled)

	glassBodyEnabled = enabled

	if enabled then
		enableGlass()
	else
		restoreGlass()
	end

end

player.CharacterAdded:Connect(function()

	task.wait(1)

	if glassBodyEnabled then
		enableGlass()
	end

end)

local function applyESP(obj, category)

	if not obj or not obj.Parent then
		return
	end

	local settings = CONFIG[category]

	if not settings then
		return
	end

	local hName = category .. "Highlight"
	local bName = category .. "Label"

	local highlight = obj:FindFirstChild(hName)

	if not highlight then

		highlight = Instance.new("Highlight")

		highlight.Name = hName

		highlight.FillColor = settings.color
		highlight.OutlineColor = settings.color

		highlight.FillTransparency = 0.5
		highlight.OutlineTransparency = 0

		highlight.DepthMode =
			Enum.HighlightDepthMode.AlwaysOnTop

		highlight.Parent = obj

	end

	local billboard = obj:FindFirstChild(bName)

	if not billboard then

		billboard = Instance.new("BillboardGui")

		billboard.Name = bName

		billboard.Size =
			UDim2.fromOffset(120, 32)

		billboard.StudsOffset =
			Vector3.new(0, 2.2, 0)

		billboard.AlwaysOnTop = true
		billboard.MaxDistance = 1000

		billboard.Parent = obj

		local text = Instance.new("TextLabel")

		text.Name = "Text"

		text.Size = UDim2.fromScale(1, 1)

		text.BackgroundTransparency = 1

		text.Text = settings.label

		text.TextColor3 = settings.color

		text.TextStrokeTransparency = 0
		text.TextStrokeColor3 = Color3.new(0, 0, 0)

		text.Font = Enum.Font.GothamBold
		text.TextScaled = true

		text.Parent = billboard

	end
end

local function removeESP(obj, category)

	if not obj then
		return
	end

	local h =
		obj:FindFirstChild(category .. "Highlight")

	local l =
		obj:FindFirstChild(category .. "Label")

	if h then
		h:Destroy()
	end

	if l then
		l:Destroy()
	end
end

local function processRoom(room)

	if not room then
		return
	end

	for category, finder in pairs(FINDERS) do

		local obj = finder(room)

		if obj then

			if CONFIG[category].enabled then
				applyESP(obj, category)
			else
				removeESP(obj, category)
			end

		end
	end

	for _, obj in ipairs(findAmethysts(room)) do

		if CONFIG.Amethyst.enabled then
			applyESP(obj, "Amethyst")
		else
			removeESP(obj, "Amethyst")
		end

	end
end

local function scanRooms()

	local folder =
		workspace:FindFirstChild("CurrentRooms")

	if not folder then
		return
	end

	for _, room in ipairs(folder:GetChildren()) do

		pcall(function()
			processRoom(room)
		end)

	end
end

task.spawn(function()

	while task.wait(1) do

		pcall(scanRooms)

	end

end)

task.spawn(function()

	local folder =
		workspace:WaitForChild("CurrentRooms", 15)

	if folder then

		folder.ChildAdded:Connect(function(room)

			task.wait(0.35)

			pcall(function()
				processRoom(room)
			end)

		end)

	end
end)

local function findMonsterFolder()

	for _, name in ipairs(MONSTER_FOLDER_NAMES) do

		local folder =
			workspace:FindFirstChild(name)

		if folder then
			return folder
		end

	end

	for _, obj in ipairs(workspace:GetDescendants()) do

		if obj:IsA("Folder")
			or obj:IsA("Model") then

			local lower =
				string.lower(obj.Name)

			if lower == "monsters"
				or lower == "entities"
				or lower == "monster" then

				return obj
			end
		end
	end

	return nil
end

local monitoredFolders = {}

local function monitorMonsterFolder(folder)

	if not folder then
		return
	end

	if monitoredFolders[folder] then
		return
	end

	monitoredFolders[folder] = true

	folder.ChildAdded:Connect(function(obj)

		if not monsterNotificationsEnabled then
			return
		end

		task.wait(0.05)

		if obj and obj.Parent == folder then

			showNotification(obj.Name)

		end
	end)
end

task.spawn(function()

	while task.wait(1) do

		local folder =
			findMonsterFolder()

		if folder then
			monitorMonsterFolder(folder)
		end

	end
end)
